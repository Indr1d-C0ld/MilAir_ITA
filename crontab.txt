# MilAir ITA — job periodici (da aggiungere al crontab dell'utente proprietario dei
# file dell'app; adattare il percorso se diverso da /var/www/html/milair_ita).
#
# NON servono piu': l'import del CSV (il logger scrive direttamente in events.db),
# la rarita' oraria e la scansione degli alert ogni 5 minuti (le lancia il logger
# dopo ogni registrazione, solo quando arrivano contatti nuovi), la rassegna RSS.
#
# "nice -n 19 ionice -c3": i giri che non hanno fretta, a priorita' minima.

# Ogni 3 ore aggiorna i NOTAM per la mappa (al minuto 25: lontano dalle :00, e una
# richiesta ogni 3 ore rispetta il Crawl-delay di notaminfo.com):
25 */3 * * * /usr/bin/php /var/www/html/milair_ita/fetch_notams.php >> /var/log/milair/notams.log 2>&1

# Ogni notte alle 00:15 calcola e pubblica la voce di Diario di IERI, e pota i dati
# di servizio di auth.db (non tocca la sintesi IA, che resta un passo manuale):
15 0 * * * /usr/bin/php /var/www/html/milair_ita/diary_build.php --publish --quiet >> /var/log/milair/diario.log 2>&1
# Per il primo popolamento retroattivo, una volta a mano (non da cron):
#   php /var/www/html/milair_ita/diary_build.php --publish --backfill=30

# Ogni notte alle 04:30 scarica gli asset visivi mancanti, uno script dopo l'altro
# (prima partivano tutti e quattro insieme ogni 6 ore). Per un contatto appena
# visto c'e' anche il pulsante 🔄 in tabella, che li prende subito.
30 4 * * * cd /var/www/html/milair_ita && for s in download_silhouettes download_photos download_drawings download_fdb_photos; do nice -n 19 ionice -c3 /usr/bin/php $s.php; done >> /var/log/milair/asset.log 2>&1

# La domenica alle 04:50 aggiorna i loghi compagnia/forza aerea (VRS OperatorFlags):
50 4 * * 0 nice -n 19 ionice -c3 /usr/bin/php /var/www/html/milair_ita/download_opflags.php >> /var/log/milair/loghi.log 2>&1
